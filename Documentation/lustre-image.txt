# FSNAME is the one root_image printed when it created the image.
export FSNAME=myfsname
# The pool is labelled as a disk vdev (root_image creates it on a loop
# device), so attach the image first; importing from the file fails.
LOOP=$(sudo losetup --find --show /var/lib/ktest/lustre_desktop.amd64)
sudo zpool import -N -f -d "$LOOP" lustre_pool
sudo zfs snapshot -r lustre_pool@xfer
sudo zfs send -p lustre_pool/$FSNAME-mgs_mdt@xfer | sudo zfs recv -u -x mountpoint rootfs/$FSNAME-mgs_mdt
sudo zfs send -p lustre_pool/$FSNAME-ost0@xfer | sudo zfs recv -u -x mountpoint rootfs/$FSNAME-ost0
sudo zfs send -p lustre_pool/$FSNAME-ost1@xfer | sudo zfs recv -u -x mountpoint rootfs/$FSNAME-ost1
sudo zpool export lustre_pool
sudo losetup -d "$LOOP"
